// Копия массива
func (a *Array) Clone() *Array {
        if a == nil {
                return NewArray()
        }

        copyData := make([]string, len(a.data), cap(a.data))
        copy(copyData, a.data)
        return &Array{data: copyData}
}

// копирование
func (a *Array) Assign(other *Array) bool {
        if a == nil || other == nil {
                return false
        }

        newData := make([]string, len(other.data), cap(other.data))
        copy(newData, other.data)
        a.data = newData
        return true
}

func (a *Array) reserve(newCapacity int) {
        if newCapacity <= cap(a.data) {
                return
        }

        newData := make([]string, len(a.data), newCapacity)
        copy(newData, a.data)
        a.data = newData
}

func (a *Array) AddToEnd(value string) bool {
        if len(a.data) == cap(a.data) {
                newCapacity := 1
                if cap(a.data) > 0 {
                        newCapacity = cap(a.data) * 2
                }
                a.reserve(newCapacity)
        }

        a.data = append(a.data, value)
        return true
}

func (a *Array) AddAtIndex(index int, value string) bool {
        if index < 0 || index > len(a.data) {

                if cap(a.data) > 0 {

        if err != nil {
                return false
        }
        defer file.Close()

        reader := bufio.NewReader(file)

        if err != nil || magic != "ARRAY_TEXT_V1" {
                return false
        }

        countLine, err := readTextLine(reader)
        if err != nil {
                return false
        }

        count, err := strconv.Atoi(countLine)
        if err != nil || count < 0 || count > maxArrayItems {
                return false
        }

        items := make([]string, 0, count)
        for i := 0; i < count; i++ {
                line, err := readTextLine(reader)
                if err != nil {
                        return false
                }

                value, err := strconv.Unquote(line)
                if err != nil || len(value) > maxStringBytes {
                        return false

                items = append(items, value)

        // После элементов не должно быть дополнительных данных.
        if _, err := readTextLine(reader); err != io.EOF {
                return false
        }

        a.data = items
        return true
}

        if err := binary.Read(file, binary.LittleEndian, &count); err != nil {
                return false

        if count > maxArrayItems {

        }

                        return false
                }
                if length > maxStringBytes {
                        return false
                }

                bytes := make([]byte, int(length))
                if _, err := io.ReadFull(file, bytes); err != nil {
                        return false
                }
                temp.data = append(temp.data, string(bytes))

func writeTextLine(w *bufio.Writer, line string) error {
        if _, err := w.WriteString(line); err != nil {
                return err
        }
        return w.WriteByte('\n')
}

func readTextLine(r *bufio.Reader) (string, error) {
        line, err := r.ReadString('\n')
        if err != nil && !(err == io.EOF && len(line) > 0) {
                return "", err
        }

        if err == nil {
                line = strings.TrimSuffix(line, "\n")
        }
        line = strings.TrimSuffix(line, "\r") // поддержка окончания строк CRLF

        return line, nil
}

// Узел AVL-дерева.
type NodeT struct {
        Key    int
        Left   *NodeT
        Right  *NodeT
        Height uint8
}

// AVLTree — объект дерева и указатель на его корень.
type AVLTree struct {
        root *NodeT
}

func NewAVLTree() *AVLTree {
        return &AVLTree{}
}

// Внутренние функции для высоты, балансировки и вращений.
func heightT(node *NodeT) uint8 {
        if node == nil {
                return 0
        }
        return node.Height
}

func balanceFactorT(node *NodeT) int {
        if node == nil {
                return 0
        }
        return int(heightT(node.Left)) - int(heightT(node.Right))
}

func fixHeightT(node *NodeT) {
        if node == nil {
                return
        }

        leftHeight := heightT(node.Left)
        rightHeight := heightT(node.Right)
        if leftHeight > rightHeight {
                node.Height = leftHeight + 1
        } else {
                node.Height = rightHeight + 1
        }
}

func rotateRightT(node *NodeT) *NodeT {
        newRoot := node.Left
        node.Left = newRoot.Right
        newRoot.Right = node

        fixHeightT(node)
        fixHeightT(newRoot)
        return newRoot
}

func rotateLeftT(node *NodeT) *NodeT {
        newRoot := node.Right
        node.Right = newRoot.Left
        newRoot.Left = node

        fixHeightT(node)
        fixHeightT(newRoot)
        return newRoot
}

func balanceT(node *NodeT) *NodeT {
        if node == nil {
                return nil
        }

        fixHeightT(node)
        balance := balanceFactorT(node)

        if balance == 2 {
                if balanceFactorT(node.Left) < 0 {
                        node.Left = rotateLeftT(node.Left)
                }
                return rotateRightT(node)
        }

        if balance == -2 {
                if balanceFactorT(node.Right) > 0 {
                        node.Right = rotateRightT(node.Right)
                }
                return rotateLeftT(node)
        }

        return node
}

// Вставка. Повторяющиеся ключи не добавляются.
func insertNodeT(node *NodeT, key int) *NodeT {
        if node == nil {
                return &NodeT{Key: key, Height: 1}
        }

        if key < node.Key {
                node.Left = insertNodeT(node.Left, key)
        } else if key > node.Key {
                node.Right = insertNodeT(node.Right, key)
        } else {
                return node
        }

        return balanceT(node)
}

func (tree *AVLTree) InsertT(key int) {
        if tree != nil {
                tree.root = insertNodeT(tree.root, key)
        }
}

// Удаление
func findMinNodeT(node *NodeT) *NodeT {
        if node == nil {
                return nil
        }
        for node.Left != nil {
                node = node.Left
        }
        return node
}

func removeMinT(node *NodeT) *NodeT {
        if node.Left == nil {
                return node.Right
        }
        node.Left = removeMinT(node.Left)
        return balanceT(node)
}

func removeNodeT(node *NodeT, key int) *NodeT {
        if node == nil {
                return nil
        }

        if key < node.Key {
                node.Left = removeNodeT(node.Left, key)
        } else if key > node.Key {
                node.Right = removeNodeT(node.Right, key)
        } else {
                left := node.Left
                right := node.Right

                if right == nil {
                        return left
                }

                minimum := findMinNodeT(right)
                minimum.Right = removeMinT(right)
                minimum.Left = left
                return balanceT(minimum)
        }

        return balanceT(node)
}

func (tree *AVLTree) RemoveT(key int) {
        if tree != nil {
                tree.root = removeNodeT(tree.root, key)
        }
}

// Поиск
func findNodeT(node *NodeT, key int) *NodeT {
        for node != nil {
                if key == node.Key {
                        return node
                }
                if key < node.Key {
                        node = node.Left
                } else {
                        node = node.Right
                }
        }
        return nil
}

func (tree *AVLTree) FindT(key int) *NodeT {
        if tree == nil {
                return nil
        }
        return findNodeT(tree.root, key)
}

func (tree *AVLTree) FindWithStepsT(key int) (*NodeT, int) {
        if tree == nil {
                return nil, 0
        }

        steps := 0
        for node := tree.root; node != nil; {
                steps++
                if key == node.Key {
                        return node, steps
                }
                if key < node.Key {
                        node = node.Left
                } else {
                        node = node.Right
                }
        }
        return nil, steps
}

        if tree == nil {

        count := 0
        nodes := []*NodeT{tree.root}

                node := nodes[len(nodes)-1]
                nodes = nodes[:len(nodes)-1]
                if node == nil {
                        continue
                }

                count++
                nodes = append(nodes, node.Left, node.Right)
        }
        return count
}

func loadTreeTextNodeT(tokens []string, position *int, depth int) (*NodeT, bool) {
        if depth > maxAVLDepth || *position >= len(tokens) {
                return nil, false
        }

        token := tokens[*position]
        *position++

        if token == "#" {
                return nil, true
        }

        key, err := strconv.Atoi(token)
        if err != nil {
                return nil, false
        }

        node := &NodeT{Key: key, Height: 1}

        left, ok := loadTreeTextNodeT(tokens, position, depth+1)
        if !ok {
                return nil, false
        }
        right, ok := loadTreeTextNodeT(tokens, position, depth+1)
        if !ok {
                return nil, false
        }

        node.Left = left
        node.Right = right
        fixHeightT(node)
        return node, true
}

        tokens := strings.Fields(string(data))
        position := 0

        loadedRoot, ok := loadTreeTextNodeT(tokens, &position, 0)
        if !ok || position != len(tokens) {
                return false
        }

        if tree != nil {
                tree.root = loadedRoot
        }
        return true
}

// AVLBIN01, затем маркеры узлов (0 — nil, 1 — узел)
// и для каждого узла int32-ключ, левое и правое поддеревья.
func writeTreeBinaryT(w io.Writer, node *NodeT, depth int) error {
        if depth > maxAVLDepth {
                return fmt.Errorf("превышена допустимая глубина дерева")
        }

        if node == nil {
                return binary.Write(w, binary.LittleEndian, uint8(0))
        }

        if err := binary.Write(w, binary.LittleEndian, uint8(1)); err != nil {
                return err
        }
        if err := binary.Write(w, binary.LittleEndian, int32(node.Key)); err != nil {
                return err
        }
        if err := writeTreeBinaryT(w, node.Left, depth+1); err != nil {
                return err
        }
        return writeTreeBinaryT(w, node.Right, depth+1)
}

                        root = tree.root

func readTreeBinaryT(r io.Reader, depth int, count *int) (*NodeT, bool) {
        if depth > maxAVLDepth || *count > 1_000_000 {
                return nil, false
        }

        var marker uint8
        if err := binary.Read(r, binary.LittleEndian, &marker); err != nil {
                return nil, false
        }
        if marker == 0 {
                return nil, true
        }
        if marker != 1 {
                return nil, false
        }

        var key int32
        if err := binary.Read(r, binary.LittleEndian, &key); err != nil {
                return nil, false
        }

        *count++
        node := &NodeT{Key: int(key), Height: 1}

        left, ok := readTreeBinaryT(r, depth+1, count)
        if !ok {
                return nil, false
        }
        right, ok := readTreeBinaryT(r, depth+1, count)
        if !ok {
                return nil, false
        }

        node.Left = left
        node.Right = right
        fixHeightT(node)
        return node, true
}

func (tree *AVLTree) LoadTreeBinary(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()

        var actualMagic [8]byte
        if _, err := io.ReadFull(file, actualMagic[:]); err != nil {
                return false
        }

        expectedMagic := [8]byte{'A', 'V', 'L', 'B', 'I', 'N', '0', '1'}
        if actualMagic != expectedMagic {
                return false
        }

        count := 0
        loadedRoot, ok := readTreeBinaryT(file, 0, &count)
        if !ok {
                return false
        }

        // Отклоняем файл, если после дерева есть лишние байты.
        var extra [1]byte
        if n, err := file.Read(extra[:]); n != 0 || err != io.EOF {
                return false
        }

        if tree != nil {
                tree.root = loadedRoot
        }
        return true
}

type HashMethod uint32

const (
        LinearProbing HashMethod = iota
        DoubleHashing
)

type slotState uint8

const (
        slotEmpty    slotState = iota //ячейка никогда не использовалась
        slotOccupied                  //в ячейке хранятся пары ключ значение
        slotDeleted                   // "надгробие"
)

const (
        maxHashItems     = 1_000_000
        maxHashKeyLength = 1_048_576
)

type hashEntry struct {
        key   string
        value any
        state slotState
}

type HashTable struct {
        entries    []hashEntry
        size       int
        method     HashMethod
        probes     uint64
        collisions uint64
}

func NewHashTable(expectedElements int, method HashMethod) *HashTable {
        if expectedElements < 0 {
                expectedElements = 0
        }
        if expectedElements > maxHashItems {
                expectedElements = maxHashItems
        }
        if method != LinearProbing && method != DoubleHashing {
                method = LinearProbing
        }

        capacity := nextPrimeHash(expectedElements*2 + 11)

        return &HashTable{
                entries: make([]hashEntry, capacity),
                method:  method,
        }
}

// Хеш-функции
// uint64 переполняется по модулю 2^64, как 64-битный size_t.
func hash1(key string) uint64 {
        hash := uint64(5381)
        for i := 0; i < len(key); i++ {
                hash = ((hash << 5) + hash) + uint64(key[i])
        }
        return hash
}

func hash2(key string) uint64 {
        var hash uint64
        for i := 0; i < len(key); i++ {
                hash = uint64(key[i]) + (hash << 6) + (hash << 16) - hash
        }
        return hash
}

func isPrimeHash(number int) bool {
        if number < 2 {
                return false
        }
        for divisor := 2; divisor <= number/divisor; divisor++ {
                if number%divisor == 0 {
                        return false
                }
        }
        return true
}

func nextPrimeHash(number int) int {
        if number < 2 {
                return 2
        }
        for !isPrimeHash(number) {
                number++
        }
        return number
}

func (table *HashTable) stepFor(key string) int {
        if table.method == LinearProbing || len(table.entries) <= 2 {
                return 1
        }
        return 1 + int(hash2(key)%uint64(len(table.entries)-1))
}

// Поиск. Для значений любого типа используй HashFind.
// Если значение должно быть int — HashFindInt.
func (table *HashTable) HashFind(key string) (any, bool) {
        if table == nil || len(table.entries) == 0 {
                return nil, false
        }

        index := int(hash1(key) % uint64(len(table.entries)))
        step := table.stepFor(key)

        for checked := 0; checked < len(table.entries); checked++ {
                table.probes++
                entry := &table.entries[index]

                if entry.state == slotEmpty {
                        return nil, false
                }
                if entry.state == slotOccupied && entry.key == key {
                        return entry.value, true
                }

                table.collisions++
                index = (index + step) % len(table.entries)
        }
        return nil, false
}

// Вставляет значение или обновляет уже существующий ключ.
func (table *HashTable) HashInsert(key string, value any) bool {
        if table == nil || len(table.entries) == 0 {
                return false
        }

        index := int(hash1(key) % uint64(len(table.entries)))
        step := table.stepFor(key)
        firstDeleted := -1

        for checked := 0; checked < len(table.entries); checked++ {
                table.probes++
                entry := &table.entries[index]

                if entry.state == slotOccupied && entry.key == key {
                        entry.value = value
                        return true
                }

                if entry.state == slotDeleted && firstDeleted == -1 {
                        firstDeleted = index
                }

                if entry.state == slotEmpty {
                        target := index
                        if firstDeleted != -1 {
                                target = firstDeleted
                        }

                        table.entries[target] = hashEntry{
                                key:   key,
                                value: value,
                                state: slotOccupied,
                        }
                        table.size++
                        return true
                }

                table.collisions++
                index = (index + step) % len(table.entries)
        }

        if firstDeleted != -1 {
                table.entries[firstDeleted] = hashEntry{
                        key:   key,
                        value: value,
                        state: slotOccupied,
                }
                table.size++
                return true
        }

        return false
}

func (table *HashTable) HashErase(key string) bool {
        if table == nil || len(table.entries) == 0 {
                return false
        }

        index := int(hash1(key) % uint64(len(table.entries)))
        step := table.stepFor(key)

        for checked := 0; checked < len(table.entries); checked++ {
                table.probes++
                entry := &table.entries[index]

                if entry.state == slotEmpty {
                        return false
                }

                if entry.state == slotOccupied && entry.key == key {
                        entry.state = slotDeleted
                        entry.value = nil
                        table.size--
                        return true
                }

                index = (index + step) % len(table.entries)
        }
        return false
}

// Обёртки для целочисленных значений.
func (table *HashTable) HashInsertInt(key string, value int) bool {
        return table.HashInsert(key, value)
}

func (table *HashTable) HashFindInt(key string) (int, bool) {
        value, found := table.HashFind(key)
        if !found {
                return 0, false
        }

        number, ok := value.(int)
        return number, ok
}

// Преобразует целочисленное значение к формату int32,
func valueAsInt32(value any) (int32, bool) {
        switch number := value.(type) {
        case int:
                return int32(number), true
        case int8:
                return int32(number), true
        case int16:
                return int32(number), true
        case int32:
                return number, true
        case int64:
                return int32(number), true
        case uint:
                return int32(number), true
        case uint8:
                return int32(number), true
        case uint16:
                return int32(number), true
        case uint32:
                return int32(number), true
        case uint64:
                return int32(number), true
        case uintptr:
                return int32(number), true
        default:
                return 0, false
        }
}

// Сохраняет таблицу
// HTBIN01, метод, количество записей, затем ключи и значения int32.
func (table *HashTable) SaveBinary(filename string) bool {
        if table == nil {
                return false
        }

        file, err := os.Create(filename)
        if err != nil {
                return false
        }

        writeErr := func() error {
                magic := [8]byte{'H', 'T', 'B', 'I', 'N', '0', '1', 0}
                if _, err := file.Write(magic[:]); err != nil {
                        return err
                }

                if err := binary.Write(file, binary.LittleEndian, uint32(table.method)); err != nil {
                        return err
                }
                if err := binary.Write(file, binary.LittleEndian, uint64(table.size)); err != nil {
                        return err
                }

                for i := range table.entries {
                        entry := &table.entries[i]
                        if entry.state != slotOccupied {
                                continue
                        }
                        if len(entry.key) > maxHashKeyLength {
                                return errInvalidHashFile
                        }

                        value, ok := valueAsInt32(entry.value)
                        if !ok {
                                return errInvalidHashFile
                        }

                        if err := binary.Write(file, binary.LittleEndian, uint64(len(entry.key))); err != nil {
                                return err
                        }
                        if _, err := io.WriteString(file, entry.key); err != nil {
                                return err
                        }
                        if err := binary.Write(file, binary.LittleEndian, value); err != nil {
                                return err
                        }
                }
                return nil
        }()

        closeErr := file.Close()
        return writeErr == nil && closeErr == nil
}

var errInvalidHashFile = &hashFileError{}

type hashFileError struct{}

func (*hashFileError) Error() string {
        return "неподдерживаемое значение или слишком длинный ключ"
}

func (table *HashTable) LoadBinary(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()

        var actualMagic [8]byte
        if _, err := io.ReadFull(file, actualMagic[:]); err != nil {
                return false
        }

        expectedMagic := [8]byte{'H', 'T', 'B', 'I', 'N', '0', '1', 0}
        if actualMagic != expectedMagic {
                return false
        }

        var savedMethod uint32
        var savedSize uint64

        if binary.Read(file, binary.LittleEndian, &savedMethod) != nil ||
                binary.Read(file, binary.LittleEndian, &savedSize) != nil {
                return false
        }
        if savedMethod > uint32(DoubleHashing) || savedSize > maxHashItems {
                return false
        }

        temporary := NewHashTable(int(savedSize), HashMethod(savedMethod))

        for i := uint64(0); i < savedSize; i++ {
                var keyLength uint64
                if binary.Read(file, binary.LittleEndian, &keyLength) != nil {
                        return false
                }
                if keyLength > maxHashKeyLength {
                        return false
                }

                keyBytes := make([]byte, int(keyLength))
                if _, err := io.ReadFull(file, keyBytes); err != nil {
                        return false
                }

                var value int32
                if binary.Read(file, binary.LittleEndian, &value) != nil {
                        return false
                }
                if !temporary.HashInsertInt(string(keyBytes), int(value)) {
                        return false
                }
        }

        // Не загружаем файл, если после данных остались лишние байты.
        var extra [1]byte
        if n, err := file.Read(extra[:]); n != 0 || err != io.EOF {
                return false
        }

        if uint64(temporary.size) != savedSize {
                return false
        }

        temporary.probes = 0
        temporary.collisions = 0
        *table = *temporary
        return true
}

// SaveText сохраняет хеш-таблицу в текстовый файл.

                }

                value, ok := valueAsInt32(entry.value)
                if !ok {

                }

                writeErr = writeTextLine(writer, strconv.Quote(entry.key))
                if writeErr == nil {

        if writeErr == nil {
                writeErr = writer.Flush()
        }
        closeErr := file.Close()

// LoadText загружает хеш-таблицу из текстового файла.
func (table *HashTable) LoadText(filename string) bool {
        if table == nil {
                return false
        }

        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()

        reader := bufio.NewReader(file)

        magic, err := readTextLine(reader)
        if err != nil || magic != "HASHTABLE_TEXT_V1" {
                return false
        }

        methodLine, err := readTextLine(reader)
        if err != nil {
                return false
        }

        methodValue, err := strconv.ParseUint(methodLine, 10, 32)
        if err != nil || methodValue > uint64(DoubleHashing) {
                return false
        }
        method := HashMethod(methodValue)

        countLine, err := readTextLine(reader)
        if err != nil {
                return false
        }

        count, err := strconv.Atoi(countLine)
        if err != nil || count < 0 || count > maxHashItems {
                return false
        }

        temporary := NewHashTable(count, method)

        for i := 0; i < count; i++ {
                keyLine, err := readTextLine(reader)
                if err != nil {
                        return false
                }

                key, err := strconv.Unquote(keyLine)
                if err != nil || len(key) > maxHashKeyLength {
                        return false
                }

                valueLine, err := readTextLine(reader)
                if err != nil {
                        return false
                }

                value, err := strconv.ParseInt(valueLine, 10, 32)
                if err != nil {
                        return false
                }

                oldSize := temporary.size
                if !temporary.HashInsertInt(key, int(value)) || temporary.size == oldSize {
                        return false // ошибка вставки или повторяющийся ключ
                }
        }

        // После записей не должно быть дополнительных данных.
        if _, err := readTextLine(reader); err != io.EOF {
                return false
        }

        temporary.probes = 0
        temporary.collisions = 0
        *table = *temporary
        return true
}

type NodeF struct {

type FList struct {

func NewFList() *FList {
        return &FList{}
}

// Проверка и размер
func (list *FList) IsEmptyF() bool {
        return list == nil || list.head == nil
}

func (list *FList) SizeF() int {
        if list == nil {
                return 0
        }

        count := 0
        for current := list.head; current != nil; current = current.Next {
                count++
        }
        return count
}

// Поиск
func (list *FList) FindNodeF(value string) *NodeF {
        if list == nil {
                return nil
        }

        for current := list.head; current != nil; current = current.Next {
                if current.Data == value {
                        return current
                }
        }
        return nil
}

func (list *FList) GetAtF(index int) *NodeF {
        if list == nil || index < 0 {
                return nil
        }

        current := list.head
        for i := 0; current != nil && i < index; i++ {
                current = current.Next
        }
        return current
}

// Добавление
func (list *FList) PushFrontF(value string) {
        node := &NodeF{
                Data: value,
                Next: list.head,
        }
        list.head = node

        if list.tail == nil {
                list.tail = node
        }
}

func (list *FList) PushBackF(value string) {
        node := &NodeF{Data: value}

        if list.tail == nil {
                list.head = node
                list.tail = node
                return
        }

        list.tail.Next = node
        list.tail = node
}

func (list *FList) InsertAfterF(target, value string) bool {
        targetNode := list.FindNodeF(target)
        if targetNode == nil {
                return false
        }

        node := &NodeF{
                Data: value,
                Next: targetNode.Next,
        }
        targetNode.Next = node

        if list.tail == targetNode {
                list.tail = node
        }
        return true
}

func (list *FList) InsertBeforeF(target, value string) bool {
        if list == nil || list.head == nil {
                return false
        }

        if list.head.Data == target {
                list.PushFrontF(value)
                return true
        }

        previous := list.head
        for previous.Next != nil && previous.Next.Data != target {
                previous = previous.Next
        }

        if previous.Next == nil {
                return false
        }

        node := &NodeF{
                Data: value,
                Next: previous.Next,
        }
        previous.Next = node
        return true
}

// Удаление
func (list *FList) PopFrontF() bool {
        if list == nil || list.head == nil {
                return false
        }

        list.head = list.head.Next
        if list.head == nil {
                list.tail = nil
        }
        return true
}

func (list *FList) PopBackF() bool {
        if list == nil || list.head == nil {
                return false
        }

        if list.head == list.tail {
                list.head = nil
                list.tail = nil
                return true
        }

        previous := list.head
        for previous.Next != list.tail {
                previous = previous.Next
        }

        previous.Next = nil
        list.tail = previous
        return true
}

func (list *FList) RemoveAfterF(target string) bool {
        targetNode := list.FindNodeF(target)
        if targetNode == nil || targetNode.Next == nil {
                return false
        }

        removed := targetNode.Next
        targetNode.Next = removed.Next

        if list.tail == removed {
                list.tail = targetNode
        }
        return true
}

func (list *FList) RemoveBeforeF(target string) bool {
        if list == nil || list.head == nil || list.head.Data == target {
                return false
        }

        // Узел перед target — это голова.
        if list.head.Next != nil && list.head.Next.Data == target {
                return list.PopFrontF()
        }

        // Ищем узел, стоящий за два шага до target.
        previousPrevious := list.head
        for previousPrevious.Next != nil &&
                previousPrevious.Next.Next != nil &&
                previousPrevious.Next.Next.Data != target {
                previousPrevious = previousPrevious.Next
        }

        if previousPrevious.Next == nil || previousPrevious.Next.Next == nil {
                return false
        }

        removed := previousPrevious.Next
        previousPrevious.Next = removed.Next
        return true
}

func (list *FList) RemoveByValueF(value string) bool {
        if list == nil || list.head == nil {
                return false
        }

        if list.head.Data == value {
                return list.PopFrontF()
        }

        previous := list.head
        for previous.Next != nil && previous.Next.Data != value {
                previous = previous.Next
        }

        if previous.Next == nil {
                return false
        }

        removed := previous.Next
        previous.Next = removed.Next

        if list.tail == removed {
                list.tail = previous
        }
        return true
}

// Поиск индекса. Если значение не найдено, возвращается -1.
func (list *FList) SearchF(value string) int {
        if list == nil {
                return -1
        }

        index := 0
        for current := list.head; current != nil; current = current.Next {
                if current.Data == value {
                        return index
                }
                index++
        }
        return -1
}

func (s *Stack) LoadStackBinary(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {

        }
        defer file.Close()

        var actualMagic [8]byte
        if _, err := io.ReadFull(file, actualMagic[:]); err != nil {

        temporary := NewStack()

                if temporary.head == nil {
                        temporary.head = node
                } else {
                        tail.next = node
                }
                tail = node
        }

        // текущий стек остаётся без изменений.
